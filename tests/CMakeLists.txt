cmake_minimum_required(VERSION 3.10)
project(cake_covariance_math_tests LANGUAGES CXX)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
find_package(OpenMP QUIET)
enable_testing()
foreach(test estimator_covariance photometric_selection)
  add_executable(${test}_math_test ${test}_math_test.cpp)
  target_compile_features(${test}_math_test PRIVATE cxx_std_17)
  target_include_directories(${test}_math_test PRIVATE ../include)
  target_link_libraries(${test}_math_test PRIVATE Eigen3::Eigen)
  if(test STREQUAL "photometric_selection" AND OpenMP_CXX_FOUND)
    target_link_libraries(${test}_math_test PRIVATE OpenMP::OpenMP_CXX)
    target_compile_definitions(${test}_math_test PRIVATE MP_EN MP_PROC_NUM=4)
  endif()
  # Keep assertions active in Release builds too.
  if(MSVC)
    target_compile_options(${test}_math_test PRIVATE /UNDEBUG)
  else()
    target_compile_options(${test}_math_test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME ${test}_math_test COMMAND ${test}_math_test)
endforeach()

add_executable(photometric_selection_benchmark photometric_selection_benchmark.cpp)
target_compile_features(photometric_selection_benchmark PRIVATE cxx_std_17)
target_include_directories(photometric_selection_benchmark PRIVATE ../include)
target_link_libraries(photometric_selection_benchmark PRIVATE Eigen3::Eigen)
if(OpenMP_CXX_FOUND)
  target_link_libraries(photometric_selection_benchmark PRIVATE OpenMP::OpenMP_CXX)
  target_compile_definitions(photometric_selection_benchmark PRIVATE MP_EN MP_PROC_NUM=4)
endif()
